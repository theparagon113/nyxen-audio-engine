cmake_minimum_required(VERSION 3.24)
project(Nyxen VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
set(NYXEN_JUCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/JUCE" CACHE PATH "JUCE 8.0.13 checkout")
if(NOT EXISTS "${NYXEN_JUCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Download JUCE 8.0.13 into third_party/JUCE (see README)")
endif()
set(JUCE_MODULES_ONLY ON CACHE BOOL "Only audio modules; no JUCE GUI build tools" FORCE)
add_subdirectory("${NYXEN_JUCE_DIR}" "${CMAKE_BINARY_DIR}/juce" EXCLUDE_FROM_ALL)
# Compile JUCE once; keep module sources out of consumer targets.
add_library(NyxenAudio STATIC src/OfflineProcessor.cpp)
target_link_libraries(NyxenAudio PRIVATE juce::juce_audio_formats PUBLIC NyxenCore)
target_include_directories(NyxenAudio PUBLIC include PRIVATE "${NYXEN_JUCE_DIR}/modules")
target_compile_definitions(NyxenAudio PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1
    JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0 JUCE_USE_FLAC=1 JUCE_USE_MP3AUDIOFORMAT=1
    JUCE_USE_OGGVORBIS=0 JUCE_USE_WINDOWS_MEDIA_FORMAT=0)
add_library(NyxenCore STATIC src/Engine.cpp src/Biquad.cpp src/ContinuousEngine.cpp)
target_include_directories(NyxenCore PUBLIC include)
add_executable(nyxen-cli src/main.cpp)
target_link_libraries(nyxen-cli PRIVATE NyxenAudio)
target_include_directories(nyxen-cli PRIVATE "${NYXEN_JUCE_DIR}/modules")
find_package(Python3 COMPONENTS Interpreter)
find_program(NYXEN_FFMPEG ffmpeg)
if(Python3_FOUND)
    target_compile_definitions(nyxen-cli PRIVATE NYXEN_PYTHON_EXECUTABLE="${Python3_EXECUTABLE}"
        NYXEN_SAFETY_SCRIPT="${CMAKE_CURRENT_SOURCE_DIR}/scripts/output_safety.py")
endif()
if(MSVC)
    target_compile_options(NyxenCore PRIVATE /W4 /permissive-)
    target_compile_options(nyxen-cli PRIVATE /W4 /permissive-)
endif()
add_executable(nyxen-stream-sim tests/stream_simulator.cpp tests/rt_allocation_probe.cpp)
target_link_libraries(nyxen-stream-sim PRIVATE NyxenCore)
if(WIN32)
    target_link_libraries(nyxen-stream-sim PRIVATE psapi advapi32)
endif()
if(BUILD_TESTING)
    add_test(NAME nyxen-continuous COMMAND nyxen-stream-sim "${CMAKE_BINARY_DIR}/continuous-artifacts")
    if(MSVC)
        # Dedicated ASan core: no JUCE, no intrusive allocation replacement, no CPU claims.
        add_executable(nyxen-stream-asan tests/stream_simulator.cpp src/Engine.cpp src/Biquad.cpp src/ContinuousEngine.cpp)
        target_include_directories(nyxen-stream-asan PRIVATE include)
        target_compile_definitions(nyxen-stream-asan PRIVATE NYXEN_STREAM_ASAN=1)
        target_compile_options(nyxen-stream-asan PRIVATE /fsanitize=address /Zi)
        target_link_options(nyxen-stream-asan PRIVATE /INCREMENTAL:NO /DEBUG)
        target_link_libraries(nyxen-stream-asan PRIVATE psapi advapi32)
        add_test(NAME nyxen-continuous-asan COMMAND nyxen-stream-asan "${CMAKE_BINARY_DIR}/continuous-asan-artifacts")
    endif()
    add_executable(nyxen-tests tests/tests.cpp)
    target_include_directories(nyxen-tests PRIVATE "${NYXEN_JUCE_DIR}/modules")
    target_link_libraries(nyxen-tests PRIVATE NyxenAudio)
    add_test(NAME nyxen-audio COMMAND nyxen-tests "${CMAKE_BINARY_DIR}/test-artifacts")
    add_executable(nyxen-dsp-tests tests/dsp_tests.cpp)
    target_link_libraries(nyxen-dsp-tests PRIVATE NyxenCore)
    add_test(NAME nyxen-dsp COMMAND nyxen-dsp-tests "${CMAKE_BINARY_DIR}/dsp-artifacts")
    add_executable(nyxen-precision-benchmark tests/precision_benchmark.cpp)
    target_include_directories(nyxen-precision-benchmark PRIVATE "${NYXEN_JUCE_DIR}/modules")
    target_link_libraries(nyxen-precision-benchmark PRIVATE NyxenAudio)
    find_package(Python3 COMPONENTS Interpreter)
    find_program(NYXEN_FFMPEG ffmpeg)
    if(Python3_FOUND)
        add_test(NAME nyxen-rt-source-audit COMMAND "${Python3_EXECUTABLE}" -B
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/rt_source_audit.py" "${CMAKE_CURRENT_SOURCE_DIR}")
        if(MSVC)
            add_test(NAME nyxen-asan-detector COMMAND "${Python3_EXECUTABLE}" -B
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/asan_positive_control.py" "$<TARGET_FILE:nyxen-stream-asan>"
                "${CMAKE_BINARY_DIR}/continuous-asan-artifacts")
        endif()
        add_test(NAME nyxen-cli-integration COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_integration.py" "$<TARGET_FILE:nyxen-cli>"
            "${CMAKE_BINARY_DIR}/cli-artifacts" "${NYXEN_FFMPEG}" "$<TARGET_FILE:nyxen-tests>")
        if(NYXEN_FFMPEG)
            add_test(NAME nyxen-evaluation COMMAND "${Python3_EXECUTABLE}"
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/evaluation_tests.py" "$<TARGET_FILE:nyxen-cli>"
                "${CMAKE_BINARY_DIR}/evaluation-artifacts" "${NYXEN_FFMPEG}")
            add_test(NAME nyxen-output-safety COMMAND "${Python3_EXECUTABLE}"
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/output_safety_tests.py" "$<TARGET_FILE:nyxen-cli>"
                "${CMAKE_BINARY_DIR}/safety-artifacts" "${NYXEN_FFMPEG}")
        else()
            message(WARNING "FFmpeg missing: loudness tests unavailable; M0.2 cannot be PASS")
        endif()
        add_test(NAME nyxen-numerical-precision COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/scripts/run_precision_benchmark.py"
            --executable "$<TARGET_FILE:nyxen-precision-benchmark>" --output-dir "${CMAKE_BINARY_DIR}/precision-artifacts")
    else()
        message(WARNING "Python missing: CLI/evaluation tests unavailable; M0.2 cannot be PASS")
    endif()
endif()
